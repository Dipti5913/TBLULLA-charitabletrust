rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Global rule: Public read access, authenticated write access
    match /{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Blog images
    match /blogs/{imageId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Event images
    match /events/{imageId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Project images
    match /projects/{imageId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Video thumbnails
    match /videos/{imageId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Annual reports (PDFs, documents)
    match /reports/{fileId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // CSR certificates and documents
    match /csr/{fileId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // NGO related files
    match /ngos/{fileId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // General uploads folder
    match /uploads/{fileId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Test folder (full access for debugging)
    match /test/{fileId} {
      allow read, write: if true;
    }
    
    // Temporary files
    match /temp/{fileId} {
      allow read, write: if request.auth != null;
    }
  }
}
