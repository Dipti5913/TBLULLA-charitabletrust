service cloud.firestore {
  match /databases/{database}/documents {
    // Allow public read access to published content
    match /blogs/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /events/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /projects/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /globalGrants/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /videos/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /annualReports/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /ngos/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /csrProposals/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /csrFocusAreas/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /csrCertificates/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /contactMessages/{document} {
      allow read: if request.auth != null;
      allow write: if true; // Allow public to submit contact forms
    }
    
    // Allow all authenticated users to read/write (for admin panel)
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
