rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Allow public read access to all files, authenticated write access
    match /{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Specific rules for blog images
    match /blogs/{imageId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Test folder for debugging (full access)
    match /test/{imageId} {
      allow read, write: if true;
    }
    
    // Other specific folders
    match /projects/{imageId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /events/{imageId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /reports/{imageId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
