rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow public read access to testimonials (client needs to read)
    match /testimonials/{document} {
      allow read: if true;
      allow write: if request.auth != null; // Only authenticated admin can write
    }
    
    // Allow public read access to events (client needs to read)
    match /events/{document} {
      allow read: if true;
      allow write: if request.auth != null; // Only authenticated admin can write
    }
    
    // Allow public read access to global grants (client needs to read)
    match /globalGrants/{document} {
      allow read: if true;
      allow write: if request.auth != null; // Only authenticated admin can write
    }
    
    // Allow public read access to projects (client needs to read)
    match /projects/{document} {
      allow read: if true;
      allow write: if request.auth != null; // Only authenticated admin can write
    }
    
    // Allow public read access to blogs (client needs to read)
    match /blogs/{document} {
      allow read: if true;
      allow write: if request.auth != null; // Only authenticated admin can write
    }
    
    // Allow public read access to videos (client needs to read)
    match /videos/{document} {
      allow read: if true;
      allow write: if request.auth != null; // Only authenticated admin can write
    }
    
    // Allow public read access to NGOs (client needs to read)
    match /ngos/{document} {
      allow read: if true;
      allow write: if request.auth != null; // Only authenticated admin can write
    }
    
    // Allow public read access to annual reports (client needs to read)
    match /annualReports/{document} {
      allow read: if true;
      allow write: if request.auth != null; // Only authenticated admin can write
    }
    
    // Allow public read/write to visitor analytics (for visitor count)
    match /analytics/siteVisitors {
      allow read, write: if true;
    }
    
    // Allow public write to contacts (for contact form submissions)
    match /contacts/{document} {
      allow read: if request.auth != null; // Only admin can read
      allow write: if true; // Anyone can submit contact form
    }
    
    // Default rule: require authentication for everything else
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
